def broken_vault():
    storage = {}
    
    def clear_memory():
        # This crashes! We are trying to replace the whole binder without permission.
        storage = {} 
        return storage
        
    def clear_memory_fixed():
        nonlocal storage
        # This works! We got permission to throw the old binder away.
        storage = {} 
        return storage
        # Alternatively, we could do this WITHOUT nonlocal:
        # storage.clear()  <-- This just empties the existing binder!
    return [clear_memory, clear_memory_fixed]
